% Paired lines are built from two top-aligned parboxes.  The font is selected
% *before* each parbox is constructed so TeX computes the same first-line baseline
% on both sides.  This avoids the vertical drop that occurs when a smaller date font
% is switched only inside an already-created parbox.
\newcommand{\pairedline}[4]{%
  \noindent
  {\bodyfont\parbox[t]{\dimexpr\textwidth-44mm\relax}{\raggedright #1}}%
  \hfill
  {\datefont\parbox[t]{42mm}{\raggedleft #2}}%
  \par
  \ifx\\#3\\\else
    \vspace{0.15pt}%
    \noindent
    {\bodyfont\parbox[t]{\dimexpr\textwidth-44mm\relax}{\raggedright #3}}%
    \hfill
    {\datefont\parbox[t]{42mm}{\raggedleft #4}}%
    \par
  \fi
}
